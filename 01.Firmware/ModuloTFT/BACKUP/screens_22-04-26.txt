static const char *mode_index_to_string(int mode)
{
    switch (mode) {
        case 0: return "Continuo";
        case 1: return "Bolo";
        case 2: return "KVO";
        case 3: return "Purga";
        case 4: return "Intermitente";
        case 5: return "Por tiempo";
        default: return "-";
    }
}

static void event_handler_cb_menu_local_modo_modo_seleccionado(lv_event_t *e) {
    lv_event_code_t event = lv_event_get_code(e);
    void *flowState = lv_event_get_user_data(e);
    lv_obj_t *obj = lv_event_get_target(e);
    
    if (event == LV_EVENT_VALUE_CHANGED) {
        int selected_btn = lv_btnmatrix_get_selected_btn(obj);
        if (selected_btn >= 0) {
            set_var_modo_seleccionado((int32_t)selected_btn);

            const char *selected_text = lv_btnmatrix_get_btn_text(obj, selected_btn);
            if (selected_text != NULL) {
                if (objects.obj17 != NULL) {
                    lv_label_set_text(objects.obj17, selected_text);
                }
                if (objects.label_modo != NULL) {
                    lv_label_set_text(objects.label_modo, selected_text);
                }
            }
        }

        e->user_data = (void *)0;
        flowPropagateValueLVGLEvent(flowState, 4, 0, e);
    }
}

void tick_screen_menu_local_parametros() {
    void *flowState = getFlowState(0, 5);
    (void)flowState;
    {
        const char *new_val = evalTextProperty(flowState, 13, 3, "Failed to evaluate Text in Label widget");
        const char *cur_val = lv_label_get_text(objects.label_modo);
        if (strcmp(new_val, cur_val) != 0) {
            tick_value_change_obj = objects.label_modo;
            lv_label_set_text(objects.label_modo, new_val);
            tick_value_change_obj = NULL;
        }
    }
    {
        bool new_val = evalBooleanProperty(flowState, 14, 3, "Failed to evaluate Checked state");
        bool cur_val = lv_obj_has_state(objects.modo_dropdown, LV_STATE_CHECKED);
        if (new_val != cur_val) {
            tick_value_change_obj = objects.modo_dropdown;
            if (new_val) {
                lv_obj_add_state(objects.modo_dropdown, LV_STATE_CHECKED);
            } else {
                lv_obj_clear_state(objects.modo_dropdown, LV_STATE_CHECKED);
            }
            tick_value_change_obj = NULL;
        }
    }
    {
        bool new_val = evalBooleanProperty(flowState, 15, 3, "Failed to evaluate Checked state");
        bool cur_val = lv_obj_has_state(objects.jeringa_dropdown, LV_STATE_CHECKED);
        if (new_val != cur_val) {
            tick_value_change_obj = objects.jeringa_dropdown;
            if (new_val) {
                lv_obj_add_state(objects.jeringa_dropdown, LV_STATE_CHECKED);
            } else {
                lv_obj_clear_state(objects.jeringa_dropdown, LV_STATE_CHECKED);
            }
            tick_value_change_obj = NULL;
        }
    }
}

void create_screens() {
    
    eez_flow_init_fonts(fonts, sizeof(fonts) / sizeof(ext_font_desc_t));

// Set default LVGL theme
    lv_disp_t *dispp = lv_disp_get_default();
    lv_theme_t *theme = lv_theme_default_init(dispp, lv_palette_main(LV_PALETTE_BLUE), lv_palette_main(LV_PALETTE_RED), true, LV_FONT_DEFAULT);
    lv_disp_set_theme(dispp, theme);
    
    // Initialize screens
    eez_flow_init_screen_names(screen_names, sizeof(screen_names) / sizeof(const char *));
    eez_flow_init_object_names(object_names, sizeof(object_names) / sizeof(const char *));
    
    // Create screens
    create_screen_main();
    create_screen_menu_principal();
    create_screen_menu_remoto();
    create_screen_menu_local_jeringa();
    create_screen_menu_local_modo();
    create_screen_menu_local_parametros();
    create_screen_menu_local_parametros_confirm();
    create_screen_menu_local_tiempo();
}
